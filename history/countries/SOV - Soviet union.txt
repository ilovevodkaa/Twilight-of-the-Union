# The Warsaw Pact allies are Moscow's subjects of a special kind, "socialist commonwealth states"
# (common/autonomous_states/TOTU_socialist_commonwealth.txt): their governments may change, they stay in the bloc.
# freedom_level: how far each has drifted from Moscow by January 1990. Autonomy needs Together for Victory; without it
# they are plain puppets. Subjects first, then the faction (as vanilla does).
if = {
	limit = { has_dlc = "Together for Victory" }
	set_autonomy = { target = DDR autonomous_state = autonomy_totu_socialist_commonwealth freedom_level = 0.15 }
	set_autonomy = { target = BUL autonomous_state = autonomy_totu_socialist_commonwealth freedom_level = 0.1 }
	set_autonomy = { target = CZE autonomous_state = autonomy_totu_socialist_commonwealth freedom_level = 0.4 }
	set_autonomy = { target = POL autonomous_state = autonomy_totu_socialist_commonwealth freedom_level = 0.45 }
	set_autonomy = { target = HUN autonomous_state = autonomy_totu_socialist_commonwealth freedom_level = 0.45 }
	set_autonomy = { target = ROM autonomous_state = autonomy_totu_socialist_commonwealth freedom_level = 0.5 }
	# in history files the else sits inside the if (as in vanilla DEN/ENG history)
	else = {
		puppet = DDR
		puppet = BUL
		puppet = CZE
		puppet = POL
		puppet = HUN
		puppet = ROM
	}
}
