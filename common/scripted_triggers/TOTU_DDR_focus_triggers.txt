# Moscow's own answer on the 2+4 talks: the USSR decides in totu_unity.30 when the scale first reaches 75 %
# (TOTU_unity_soviet_consent / TOTU_unity_soviet_veto), unless Moscow has nothing left to decide (no USSR, or the GDR
# is no longer its subject). While the USSR is still deciding (TOTU_unity_2p4_pending) neither flag is set.

	custom_trigger_tooltip = {
		tooltip = DDR_pol_soviet_consent_tt

			has_global_flag = TOTU_unity_soviet_consent
			NOT = { country_exists = SOV }
			NOT = { is_subject_of = SOV }
		}
	}
	custom_trigger_tooltip = {
		tooltip = DDR_pol_no_soviet_veto_tt
		NOT = { has_global_flag = TOTU_unity_soviet_veto }
	}
}
